def generate_recommendations(age, gender, occupation, diet_preference, city, allergies):

    occupation = occupation.lower()



    # Detect activity level based on keywords in occupation

    if any(word in occupation for word in ['office', 'desk', 'manager', 'it', 'software', 'admin']):

        activity_level = 'low'

    elif any(word in occupation for word in ['construction', 'labor', 'manual', 'driver', 'farmer', 'delivery']):

        activity_level = 'high'

    else:

        activity_level = 'medium'



    meals = get_meals_based_on_activity(

        diet_preference, activity_level, allergies)

    exercises = get_exercises_based_on_activity_level(activity_level)



    return {

        'meals': meals,

        'exercises': exercises

    }

import pymysql
from datetime import datetime

def insert_user(name, age, email, diet_pref, occupation=None, city=None, allergies=None, notes=None, gender=None):
    print(">>> Entering insert_user function (using PyMySQL)...")
    conn = None
    cursor = None
    try:
        print(">>> Inside try block")
        conn = pymysql.connect(
            host='localhost',
            user='root',
            password='',
            database='smartdiet',
            port=3306,
            charset='utf8mb4',
            cursorclass=pymysql.cursors.DictCursor
        )
        print(">>> Database connection successful (PyMySQL)")
        cursor = conn.cursor()
        print(">>> Cursor created (PyMySQL)")
        query = """
        INSERT INTO user_profiles (name, age, email, diet_pref, occupation, city, allergies, notes, submitted_at, gender)
        VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        """
        values = (name, age, email, diet_pref, occupation, city, allergies, notes, datetime.now(), gender)
        print(">>> SQL query defined:", query)
        print(">>> Values to insert:", values)

        print(">>> About to execute query...")
        cursor.execute(query, values)
        print(">>> Query executed (PyMySQL)")

        print(">>> About to commit changes...")
        conn.commit()
        print(">>> Changes committed (PyMySQL)")

        print(f"User '{name}' data inserted successfully (PyMySQL).")
        print(">>> Exiting try block successfully")

    except pymysql.Error as err:
        print(f">>> PyMySQL Error: {err}")
        if conn:
            conn.rollback()
            print(">>> Transaction rolled back due to error")

    finally:
        print(">>> Entering finally block")
        if cursor:
            cursor.close()
            print(">>> Cursor closed")
        if conn:
            conn.close()
            print(">>> Database connection closed")
        print(">>> Exiting finally block")

    print(">>> Exiting insert_user function")
